from dal.dal_factory import RatesFactory

class InputBll:
    def __init__(self, data_source: str):
        self.rate_dao = RatesFactory().create_instance(data_source)
        data = self.rate_dao.get_rates()
        self.valid_currencies = data.get("currencies", [])

    def get_menu_choice(self, valid_choices: list[int]):
        val = input("Option: ").strip()
        if not val.isdigit():
            raise ValueError("[Error] Invalid input. Enter a number.")
        choice = int(val)
        if choice not in valid_choices:
            raise ValueError(f"[Warn] Invalid option. Choose from {valid_choices}.")
        return choice

    def get_currency(self, label: str):
        val = input(f"{label}: ").strip().upper()
        if val not in self.valid_currencies:
            valid_list = ", ".join(self.valid_currencies)
            raise ValueError(f"[Error] Invalid currency '{val}'. [Info] Valid: {valid_list}.")
        return val

    def get_amount(self, label: str):
        val = input(f"{label}: ").strip()
        try:
            num = float(val)
            if num <= 0:
                raise ValueError("[Error] Amount must be greater than zero.")
            return num
        except ValueError:
            raise ValueError("[Error] Invalid amount. [Info] Please enter a number.")

from dal.abstract_rates import RateABC
from dal.dal_factory import RatesFactory

class RatesBll:
    __rate_dao: RateABC

    def __init__(self, data_source: str):
        self.__rate_dao = RatesFactory().create_instance(data_source)

    def get_rates_data(self):
        return self.__rate_dao.get_rates()

    def convert_amount(self, source: str, target: str, amount: float):
        """Perform direct peer-to-peer conversion using rates data."""
        source = source.upper()
        target = target.upper()
        get_rate = self.__rate_dao.get_currency_rate

        try:
            rate = get_rate(source, target)  
            return round(float(amount) * rate, 2)

        except KeyError:
            raise ValueError(
                f"[Error] Conversion rate for {source}-{target} not found. "
                f"[Info] Please verify your currency rates list."
            )

        except ZeroDivisionError:
            raise ValueError("[Error] Conversion rate invalid. [Warn] Division by zero aborted.")

        except Exception as e:
            raise ValueError(f"[Error] Conversion failed. [Info] {str(e)}")



from util.file_util import read_json_as_dict
from dal.abstract_rates import RateABC

class RateDao(RateABC):
    def __init__(self):
        self.data = read_json_as_dict("rates.json")

    def get_rates(self):
        return self.data

    def get_currency_rate(self, source: str, target: str):
        pairs = self.data.get("rates", {})   
        key = f"{source}-{target}"          
        if key in pairs:
            return pairs[key]

        rev_key = f"{target}-{source}"      
        if rev_key in pairs:
            return 1 / pairs[rev_key]

        raise KeyError(f"{source}-{target}")
